" + "Seu pedido está pendente de pagamento."; document.getElementById( "metodoSaque" ).value = ""; document.getElementById( "chaveSaque" ).value = ""; }catch(erro){ console.error(erro); msg.textContent = "❌ Erro de conexão. Tente novamente."; } }
